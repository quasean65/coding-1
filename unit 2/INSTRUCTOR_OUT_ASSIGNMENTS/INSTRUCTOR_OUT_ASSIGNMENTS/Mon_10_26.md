import random

def play_rock_paper_scissors():
    # Map raw input characters to full semantic names for output rendering
    choices_map = {'r': 'Rock', 'p': 'Paper', 's': 'Scissors'}
    valid_actions = list(choices_map.keys())
    
    # Initialize state accumulators
    player_score = 0
    cpu_score = 0
    current_round = 1
    max_rounds = 3

    print("=== Rock, Paper, Scissors Game Initialized ===")
    print("Controls: [r] Rock, [p] Paper, [s] Scissors, [q] Quit\n")

    # Main game execution loop constrained to exactly 3 successfully played rounds
    while current_round <= max_rounds:
        print(f"--- Round {current_round} of {max_rounds} ---")
        user_input = input("Enter your selection: ").strip().lower()

        # Immediate interrupt routine check
        if user_input == 'q':
            print("\nTermination signal received. Exiting game loop immediately.")
            break

        # Input sanitization and validation boundary
        if user_input not in valid_actions:
            print("Invalid input sequence. Please enter 'r', 'p', 's', or 'q'.\n")
            continue

        # Execute CPU randomized determination step
        cpu_input = random.choice(valid_actions)
        
        player_choice = choices_map[user_input]
        cpu_choice = choices_map[cpu_input]
        
        print(f"Player selected: {player_choice}")
        print(f"CPU selected:    {cpu_choice}")

        # Evaluate logic metrics for state resolution
        if user_input == cpu_input:
            print("Round Result: It's a tie!")
        elif (user_input == 'r' and cpu_input == 's') or \
             (user_input == 'p' and cpu_input == 'r') or \
             (user_input == 's' and cpu_input == 'p'):
            print("Round Result: Player wins this round!")
            player_score += 1
        else:
            print("Round Result: CPU wins this round!")
            cpu_score += 1
